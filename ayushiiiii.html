let current = 0;


function displayPage() {

const p = pages[current];


document.getElementById("emoji").innerText = p.emoji;

document.getElementById("title").innerText = p.title;

document.getElementById("message").innerText = p.message;

document.getElementById("decoration").innerText = p.decoration;


document.getElementById("counter").innerText =

String(current + 1).padStart(2,"0") +

" / " + pages.length;


document.getElementById("progress").style.width =

((current + 1) / pages.length * 100) + "%";


document.getElementById("back").style.display =

current === 0 ? "none" : "inline-block";


document.getElementById("next").style.display =

current === pages.length - 1 ? "none" : "inline-block";


document.getElementById("next").innerText =

current === 0 ? "OPEN MY LITTLE GIFT ♡" :

current === pages.length - 2 ? "ONE LAST THING... 💗" :

"NEXT PAGE ♡";


document.getElementById("replay").style.display =

current === pages.length - 1 ? "inline-block" : "none";


// restart the entrance animation

const card = document.querySelector(".card");

card.style.animation = "none";

void card.offsetWidth;

card.style.animation = "appear .45s ease";

}


function nextPage() {

if (current < pages.length - 1) {

current++;

displayPage();

}

}


function previousPage() {

if (current > 0) {

current--;

displayPage();

}

}


function restart() {

current = 0;

displayPage();

}


const style = document.createElement("style");

style.textContent = `

@keyframes appear {

from { opacity: 0; transform: translateY(12px); }

to { opacity: 1; transform: translateY(0); }

}`;

document.head.appendChild(style);


displayPage();


// Floating decorations

const floating = document.getElementById("floating");

const symbols = ["💗","💕","♡","🌸","🎀","✨","🌷","💖"];


function createFloating() {

const item = document.createElement("span");

item.className = "float";

item.innerText = symbols[Math.floor(Math.random()*symbols.length)];


item.style.left = Math.random()*100 + "vw";

item.style.fontSize = (14 + Math.random()*20) + "px";


const duration = 6 + Math.random()*5;

item.style.animationDuration = duration + "s";


floating.appendChild(item);


setTimeout(() => item.remove(), duration*1000);

}


setInterval(createFloating, 350);
